1.
Create a Departments table with:
* department_id as the Primary Key
* department_name
Then create an Employees table where:
* employee_id is the Primary Key
* department_id references the Departments table.
Create table departments(department_id int primary key, department_name varchar(50));
Create table employees(employees_id int primary key, foreign key(employees_id) references departments(department_id));
